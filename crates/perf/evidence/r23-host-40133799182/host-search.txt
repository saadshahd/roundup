name: build-issue

# L23: one build job and its own `after`, called once per Issue by build.yml's matrix. Each Issue's `after` runs as
# soon as its build ends, so a finished job frees its slot and starts the next pick without waiting for the other
# Issues of its run. `after` holds the write token the Builder never sees.
on:
  workflow_call:
    inputs:
      issue:
        required: true
        type: number
      slug:
        required: true
        type: string

permissions:
  contents: read

# L98: bash -eo pipefail, so a failure left of a pipe fails the step.
defaults:
  run:
    shell: bash

jobs:
  build:
    # A second job for one Issue waits, then finds the first one's PR and starts no run.
    concurrency:
      group: build-${{ inputs.issue }}
      cancel-in-progress: false
    runs-on: ${{ fromJSON(contains(fromJSON(vars.BOXD_ISSUES || '[]'), inputs.issue) && format('["self-hosted","boxd","roundup-build-{0}-{1}-{2}"]', inputs.issue, github.run_id, github.run_attempt) || '["ubuntu-latest"]') }}
    timeout-minutes: 120
    permissions:
      contents: read
      issues: read
      pull-requests: read
      id-token: write
    outputs:
      ran: ${{ steps.worked.outputs.ran }}
    env:
      ISSUE: ${{ inputs.issue }}
